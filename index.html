#!/bin/bash
set -e

echo "=== CLEAN SLATE Hadoop setup START ==="

# ----------------------------
# 1. Delete old files (if exist)
# ----------------------------
sudo rm -f /usr/local/bin/hadoop-hosts-setup.sh
sudo rm -f /etc/systemd/system/hadoop-hosts.service

echo "✔ Old files removed"

# ----------------------------
# 2. Create fresh host script
# ----------------------------
sudo tee /usr/local/bin/hadoop-hosts-setup.sh > /dev/null <<'EOF'
#!/bin/bash

HOSTS_FILE="/etc/hosts"

# you can list more hosts here
HOSTS=(
[host-lists]
)

# Remove all old entries dynamically
for entry in "${HOSTS[@]}"; do
    ip=$(echo "$entry" | awk '{print $1}')
    hostname=$(echo "$entry" | awk '{print $2}')
    sudo sed -i "/$ip $hostname/d" "$HOSTS_FILE"
done

# Append fresh clean block (single source of truth)
{
for entry in "${HOSTS[@]}"; do
    echo "$entry"
done
} | sudo tee -a "$HOSTS_FILE" > /dev/null

EOF

sudo chmod +x /usr/local/bin/hadoop-hosts-setup.sh

echo "✔ New host script created"

# ----------------------------
# 3. Create fresh systemd service
# ----------------------------
sudo tee /etc/systemd/system/hadoop-hosts.service > /dev/null <<EOF
[Unit]
Description=Hadoop Hosts Setup (Clean Slate)
After=network.target

[Service]
Type=oneshot
ExecStart=/usr/local/bin/hadoop-hosts-setup.sh

[Install]
WantedBy=multi-user.target
EOF

echo "✔ New systemd service created"

# ----------------------------
# 4. Reload systemd + enable
# ----------------------------
sudo systemctl daemon-reload
sudo systemctl enable hadoop-hosts.service
sudo systemctl restart hadoop-hosts.service

echo "=== CLEAN SLATE Hadoop setup DONE ==="

#!/bin/bash
RED='\033[0;31m'          # Red
GREEN='\033[0;32m'        # Green
YELLOW='\033[0;33m'       # Yellow
RM_BOLD='\033[0m'        # Remove Bold
NC='\033[0m' # No Color
PEER_IP="10.0.0.1"
_VPN_SERVER_ADDR="[vpn-server-ip]"
_VPN_CLIENT_NAME="rspi58gb"

sudo apt-get update
sudo apt-get install -y wireguard

if ping -c 1 -W 1 "$PEER_IP" >/dev/null 2>&1; then
        echo -e "${GREEN}✓$(date) - VPN OK:  reachable${NC}"
else
        echo -e "${YELLOW}$(date) - VPN is Down: Trying to establish the connection with $PEER_IP.${NC}"
        wg genkey | tee /home/ubuntu/client_private.key | wg pubkey > /home/ubuntu/client_public.key
        sudo chmod 400 /home/ubuntu/QM.pem
        conf_fileName="${_VPN_CLIENT_NAME}.conf"
cat <<EOF > /home/ubuntu/$conf_fileName
[Interface]
PrivateKey = $(sudo ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -i /home/ubuntu/QM.pem ubuntu@${_VPN_SERVER_ADDR} sudo cat /home/ubuntu/server_private.key | xargs)
Address = 10.0.0.1/24
ListenPort = 51820
MTU = 1280

[Peer]
PublicKey = $(sudo cat /home/ubuntu/client_public.key | xargs)
AllowedIPs = 10.0.0.2/32
EOF
sudo scp -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -i /home/ubuntu/QM.pem $conf_fileName ubuntu@${_VPN_SERVER_ADDR}:/tmp/
sudo ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -i /home/ubuntu/QM.pem ubuntu@${_VPN_SERVER_ADDR} \
"sudo mv /tmp/$conf_fileName /etc/wireguard/ && sudo chown root:root /etc/wireguard/$conf_fileName && sudo chmod 600 /etc/wireguard/$conf_fileName"
sudo ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -i /home/ubuntu/QM.pem ubuntu@${_VPN_SERVER_ADDR} "sudo systemctl daemon-reload"
sudo ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -i /home/ubuntu/QM.pem ubuntu@${_VPN_SERVER_ADDR} "sudo systemctl enable wg-quick@$_VPN_CLIENT_NAME"
sudo ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -i /home/ubuntu/QM.pem ubuntu@${_VPN_SERVER_ADDR} "sudo systemctl restart wg-quick@$_VPN_CLIENT_NAME"
sudo ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -i /home/ubuntu/QM.pem ubuntu@${_VPN_SERVER_ADDR} "sudo sysctl -w net.ipv4.ip_forward=1
echo 'net.ipv4.ip_forward=1' | sudo tee -a /etc/sysctl.conf
sudo sysctl -p

# for handshaking of Masquerade all VPN tunnel traffic
sudo iptables -t nat -A POSTROUTING -o ${_VPN_CLIENT_NAME} -j MASQUERADE

# Forward TCP ports except 22 (SSH) and 88 (python server) to vpn
sudo iptables -t nat -A PREROUTING -i ens5 -p tcp -m multiport ! --dports [vpn-server-port-not-to-forward] -j DNAT --to-destination 10.0.0.2

# Forward UDP ports except 51820 (WireGuard) to vpn
sudo iptables -t nat -A PREROUTING -i ens5 -p udp ! --dport 51820 -j DNAT --to-destination 10.0.0.2

# alloq all Masquerade return traffic from vpn to browser
sudo iptables -t nat -A POSTROUTING -s 10.0.0.2/32 -o ens5 -j MASQUERADE"
sudo cat <<EOF > /tmp/vpn_client.conf
[Interface]
PrivateKey = $(sudo cat /home/ubuntu/client_private.key | xargs)
Address = 10.0.0.2/24
MTU = 1280

[Peer]
# EC2
PublicKey = $(sudo ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -i /home/ubuntu/QM.pem ubuntu@${_VPN_SERVER_ADDR} sudo cat /home/ubuntu/server_public.key | xargs)
Endpoint = ${_VPN_SERVER_ADDR}:51820
AllowedIPs = 10.0.0.0/24, 192.168.1.0/24, fd00::/64
PersistentKeepalive = 25
EOF
sudo mv /tmp/vpn_client.conf /etc/wireguard/ && sudo chown root:root /etc/wireguard/vpn_client.conf && sudo chmod 600 /etc/wireguard/vpn_client.conf
sudo systemctl daemon-reload
sudo systemctl enable wg-quick@vpn_client
sudo systemctl restart wg-quick@vpn_client
fi
